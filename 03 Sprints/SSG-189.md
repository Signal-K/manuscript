```py
import math

def planet_density(R_star_solar, flux_drop):
	K = 0.414 #mass-radius empirical constant for gas giants
	alpha = 2.06 # mass-radius exponent
	pi = math.pi

	# unit conversion constant
	M_earth = 5.972e24 # kg
	R_earth = 6.371e6 # m
	R_sun = 6.957e8 #m
	unit_convert = M_earth / (R_earth**3 * R_sun**3)

	# radius of planet (in solar radii)
	Rp_solar = R_star_solar * math.sqrt(flux_drop)

	# planet mass (Earth masses)
	mass_earth = K * (Rp_solar * R_sun / R_earth)**alpha

	# Volume in m^3
	radius_m = Rp_solar * R_sun
	volume = (4/3) * pi * radius_m**3

	# mass in kg
	mass_kg = mass_earth * M_earth

	# density in kg/m^3
	density = mass_kg / volume
	return density
```
